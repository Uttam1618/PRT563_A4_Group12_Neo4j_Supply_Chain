// ============================================================================
//  PRT563 Advanced Data Management, Assessment 4, Group 12
//  importA.txt : the NAIVE graph model (Model A), built only for the
//                Model A versus Model B performance experiment
//
//  Rules applied mechanically: one label per table (29), one node per row,
//  every column a property (foreign key columns kept), one relationship type
//  per foreign key (38), named CHILDTABLE_PARENTTABLE, pointing child to parent.
//  Result   : 16,786 nodes, 31,740 relationships
//
//  Run in a SEPARATE database so that Model B (database neo4j) is untouched:
//    1. Choose the 'system' database in the Database menu and run
//         CREATE DATABASE modela IF NOT EXISTS
//    2. Choose 'modela' in the Database menu, then run this file.
//  It reads the same 29 CSV files from the instance import folder.
// ============================================================================

// ---- PART 1  Constraints and indexes on every table key ---------------------

CREATE CONSTRAINT a_ordering_officer IF NOT EXISTS FOR (n:OrderingOfficer) REQUIRE n.employee_id IS UNIQUE;
CREATE CONSTRAINT a_replenishment_assistant IF NOT EXISTS FOR (n:ReplenishmentAssistant) REQUIRE n.employee_id IS UNIQUE;

CREATE CONSTRAINT a_delivery IF NOT EXISTS FOR (n:Delivery) REQUIRE n.delivery_id IS UNIQUE;
CREATE INDEX a_delivery_line IF NOT EXISTS FOR (n:DeliveryLine) ON (n.delivery_id, n.line_no);
